// ========== ИСПРАВЛЕННАЯ ФУНКЦИЯ loadServers ========== function loadServers() { onValue(ref(db, 'publicServers'), async (snap) => { let publicServers = []; if (snap.exists()) { snap.forEach((child) => { publicServers.push({ id: child.key, ...child.val(), type: 'public', name: 'Публичный #' + child.key.slice(0, 4) }); }); } const vipSnap = await get(ref(db, 'vipServers')); let vipServers = []; if (vipSnap.exists()) { vipSnap.forEach((child) => { vipServers.push({ id: child.key, ...child.val(), type: 'vip' }); }); } const all = [...vipServers, ...publicServers]; serverList.innerHTML = ''; if (all.length === 0) { serverList.innerHTML = '
Нет серверов
'; return; } all.forEach(s => { const isVip = s.type === 'vip'; const div = document.createElement('div'); div.className = 'server-item'; // --- Безопасное создание элементов через textContent --- const infoDiv = document.createElement('div'); const nameSpan = document.createElement('strong'); nameSpan.textContent = s.name || 'Сервер'; infoDiv.appendChild(nameSpan); const badgeSpan = document.createElement('span'); badgeSpan.className = isVip ? 'badge-vip' : 'badge-public'; badgeSpan.textContent = isVip ? 'VIP' : 'Публичный'; infoDiv.appendChild(badgeSpan); const playersDiv = document.createElement('div'); playersDiv.style.fontSize = '13px'; playersDiv.style.color = '#aaa'; playersDiv.textContent = `${s.players || 0}/${s.max || 8} игроков`; infoDiv.appendChild(playersDiv); if (isVip && s.owner) { const ownerDiv = document.createElement('div'); ownerDiv.style.fontSize = '12px'; ownerDiv.style.color = '#f7c948'; ownerDiv.textContent = `Владелец: ${s.owner}`; infoDiv.appendChild(ownerDiv); } const btn = document.createElement('button'); btn.className = 'join-btn'; btn.textContent = 'Join'; // Обработчик клика btn.addEventListener('click', () => joinServer(s.id, isVip)); div.appendChild(infoDiv); div.appendChild(btn); serverList.appendChild(div); }); }); }
